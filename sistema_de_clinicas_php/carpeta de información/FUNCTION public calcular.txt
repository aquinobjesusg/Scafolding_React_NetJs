-- FUNCTION: public.calcular_edad(integer)

DROP FUNCTION IF EXISTS public.calcular_edad(integer);

CREATE OR REPLACE FUNCTION public.calcular_edad(
    p_id_usuario integer DEFAULT 0,
    OUT v_edad integer)
RETURNS integer
LANGUAGE sql
STABLE
PARALLEL RESTRICTED
AS $$
    -- Calcula la edad en años a partir de la fecha de nacimiento.
    -- Si no existe el usuario, devuelve NULL (mejor que 0).
   SELECT COALESCE(
    (SELECT EXTRACT(YEAR FROM age(CURRENT_DATE, fecha_nacimiento))::integer
     FROM public.usuarios WHERE id_usuario = p_id_usuario),
    0
);
$$;
ALTER FUNCTION public.calcular_edad(integer)
    OWNER TO postgres;
